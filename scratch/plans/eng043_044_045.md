```sql
         sum(sales_incl_vat - vat_value) filter (where sale_date between date '2026-07-01' and date '2026-07-07') as exvat_before
  from sigma_sales
  where period_kind = 'T' and txn_kind = 1 and sale_date between date '2026-07-01' and date '2026-07-26'
  group by store_code, product_code),
alive as (
  select s.*,
         (coalesce(q_gap, 0) = 0 and not exists (
            select 1 from sigma_movements m
            where m.store_code = s.store_code and m.product_code = s.product_code
              and m.movement_type = 'K' and m.movement_date between date '2026-07-08' and date '2026-07-22')) as silent
  from s where q_before > 0 and q_after > 0)
select store_code, count(*) as alive_both_sides,
       count(*) filter (where silent) as silent_both_channels,
       round(100.0 * count(*) filter (where silent) / count(*), 1) as pct_silent,
       round(sum(q_before / 7.0 * 15) filter (where silent)) as est_units_lost_at_pre_gap_rate,
       round(sum(exvat_before / 7.0 * 15) filter (where silent)) as est_exvat_rand_lost
from alive group by store_code order by store_code;
```
The two estimate columns are ESTIMATES at the seven-day pre-gap rate. Some silent lines are legitimate lulls, so read the totals as an upper-leaning figure and say so.

### Q14. ENG-043: is any DC dark today?
```sql
with r as (
  select m.store_code, m.movement_date
  from sigma_movements m
  join v_supplier_class sc on sc.store_code = m.store_code and sc.supplier_nr = m.supplier_nr
  where m.movement_type in ('R','W') and m.qty > 0 and sc.supplier_type = 'Z'
    and m.movement_date >= current_date - 182
  group by m.store_code, m.movement_date),
g as (
  select store_code, movement_date,
         movement_date - lag(movement_date) over (partition by store_code order by movement_date) as gap
  from r)
select store_code, max(movement_date) as last_dc_receipt, current_date - max(movement_date) as days_since,
       percentile_disc(0.5) within group (order by gap) as median_gap_days,
       max(gap) as longest_gap_days,
       (select max(value_num) from forge_config where config_key = 'regime_outlier_gap_multiple' and retired_on is null) as regime_multiple
from g group by store_code order by store_code;
```
Flag any store where `days_since` exceeds `regime_multiple` times `median_gap_days`. Known: TOPS Dice 80579 was dry in July for its own reasons (`SB-CC-BLOOM-017_ordering-rebuild.md:288`).
